def now(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def save(name, obj):
    with (OUT / (name+'.json')).open('x') as f: json.dump(obj,f,indent=2,ensure_ascii=False); f.write('\n')
def load(name): return json.loads((OUT/(name+'.json')).read_text())
def run(name, argv, env=None, cwd=ROOT, timeout=25):
    start=time.monotonic(); r={'argv':argv,'cwd':str(cwd),'started_at':now()}
    try:
        p=subprocess.run(argv,env=env,cwd=cwd,stdin=subprocess.DEVNULL,capture_output=True,text=True,timeout=timeout)
        r.update(exit_code=p.returncode,stdout=p.stdout,stderr=p.stderr)
    except subprocess.TimeoutExpired as e:
        r.update(exit_code=None,timed_out=True,stdout=(e.stdout or b'').decode() if isinstance(e.stdout,bytes) else e.stdout,stderr=(e.stderr or b'').decode() if isinstance(e.stderr,bytes) else e.stderr)
    except OSError as e: r.update(exit_code=None,error=str(e))
    r['duration_ms']=round((time.monotonic()-start)*1000); save(name,r)
    print(json.dumps({'evidence':name,'exit_code':r.get('exit_code'),'stdout_preview':r.get('stdout','')[:1600],'stderr':r.get('stderr','')},ensure_ascii=False)); return r
def cmux(name,*args): return run(name,[CMUX,'--json','--id-format','uuids',*args])

    env={k:os.environ[k] for k in allowed if k in os.environ}; env.update(overrides)

def screen(name,tool):
    r=cmux(name,'read-screen',*target(tool),'--scrollback','--lines','120')
    if r.get('exit_code')==0:
        obj=json.loads(r['stdout']); text=obj.get('text')
        if text is None: text=base64.b64decode(obj['base64']).decode(errors='replace')
        save(name+'-decoded',{'source':name+'.json','text':text}); print(text)
def api(name,method,path,body=None):
    url='http://127.0.0.1:19731'+path; r={'method':method,'url':url,'started_at':now(),'request':body}
    try:
        req=urllib.request.Request(url,data=json.dumps(body).encode() if body is not None else None,headers={'Content-Type':'application/json'},method=method)
        with urllib.request.urlopen(req,timeout=20) as p: r.update(status=p.status,body=p.read().decode())
    except Exception as e:
        r['error']=str(e)
        if hasattr(e,'read'): r['body']=e.read().decode()
    save(name,r); print(json.dumps(r,ensure_ascii=False)); return r
